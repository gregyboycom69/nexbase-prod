=== SQL TO RUN IN SUPABASE ===

CREATE TABLE IF NOT EXISTS workspace_tables (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE,
  name TEXT NOT NULL,
  slug TEXT NOT NULL,
  fields JSONB DEFAULT '[]',
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(workspace_id, slug)
);

ALTER TABLE workspace_tables ENABLE ROW LEVEL SECURITY;

DO $$ BEGIN
  IF NOT EXISTS (
    SELECT FROM pg_policies WHERE tablename='workspace_tables' AND policyname='owner_wt'
  ) THEN
    CREATE POLICY owner_wt ON workspace_tables FOR ALL USING (
      workspace_id IN (SELECT id FROM workspaces WHERE owner_id = auth.uid())
    );
  END IF;
END $$;

CREATE TABLE IF NOT EXISTS workspace_queries (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE,
  name TEXT NOT NULL,
  definition JSONB DEFAULT '{}',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE workspace_queries ENABLE ROW LEVEL SECURITY;

DO $$ BEGIN
  IF NOT EXISTS (
    SELECT FROM pg_policies WHERE tablename='workspace_queries' AND policyname='owner_wq'
  ) THEN
    CREATE POLICY owner_wq ON workspace_queries FOR ALL USING (
      workspace_id IN (SELECT id FROM workspaces WHERE owner_id = auth.uid())
    );
  END IF;
END $$;
