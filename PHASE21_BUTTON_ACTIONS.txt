FIX 21.1 - ADD ACTION PROPERTY TO BUTTON CONTROL
═══════════════════════════════════════════════════════════════════

In the Property Sheet, when a Button is selected, show an 
"Action" dropdown with these options:

  - none (default)
  - save
  - new
  - delete
  - first
  - previous
  - next
  - last
  - refresh

Find where Button properties are configured in:
src/components/PropertySheet.tsx (or similar)

Add this property field:

<PropRow 
  label="Action" 
  value={ctrl.props?.action || 'none'} 
  onChange={(v) => onUpdate('action', v)}
  type="select"
  options={[
    { value: 'none', label: 'No action' },
    { value: 'save', label: 'Save Record' },
    { value: 'new', label: 'New Record' },
    { value: 'delete', label: 'Delete Record' },
    { value: 'first', label: 'First Record' },
    { value: 'previous', label: 'Previous Record' },
    { value: 'next', label: 'Next Record' },
    { value: 'last', label: 'Last Record' },
    { value: 'refresh', label: 'Refresh Data' },
  ]}
/>

Also add a "Label" property if not present:
<PropRow 
  label="Label" 
  value={ctrl.props?.label || 'Button'} 
  onChange={(v) => onUpdate('label', v)}
  type="text"
/>

FIX 21.2 - WIRE BUTTON ACTIONS IN PUBLISHED APP
═══════════════════════════════════════════════════════════════════

In src/app/app/[slug]/page.tsx

The handleButtonClick function should handle all 9 action types.
Look around line 250-270 where this exists.

Update the function to handle all actions:

const handleButtonClick = async (ctrl: any) => {
  const action = ctrl.props?.action || 'none'
  
  if (action === 'none') {
    showToast('This button has no action. Set Action property in designer.', 'warning')
    return
  }
  
  if (action === 'save') {
    await handleSaveAction()
  } else if (action === 'new') {
    handleNewAction()
  } else if (action === 'delete') {
    await handleDeleteAction()
  } else if (action === 'first') {
    setCurrentRecordIndex(0)
    loadRecordIntoForm(0)
  } else if (action === 'previous') {
    if (currentRecordIndex > 0) {
      const newIndex = currentRecordIndex - 1
      setCurrentRecordIndex(newIndex)
      loadRecordIntoForm(newIndex)
    }
  } else if (action === 'next') {
    if (currentRecordIndex < tableData.length - 1) {
      const newIndex = currentRecordIndex + 1
      setCurrentRecordIndex(newIndex)
      loadRecordIntoForm(newIndex)
    }
  } else if (action === 'last') {
    const lastIndex = tableData.length - 1
    if (lastIndex >= 0) {
      setCurrentRecordIndex(lastIndex)
      loadRecordIntoForm(lastIndex)
    }
  } else if (action === 'refresh') {
    await loadPageData(activePageId!)
    showToast('Data refreshed', 'success')
  }
}

Make sure loadRecordIntoForm function exists - it loads a 
record's data into the form fields for editing.
